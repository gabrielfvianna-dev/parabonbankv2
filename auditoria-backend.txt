const mysql = require("mysql2/promise");

const config = require("./config");

// Pool de conexões com o MySQL.
// dateStrings: as datas chegam como texto ("2026-10-19 15:07:00"),
// evitando conversões de fuso horário.
// decimalNumbers fica desligado: valores DECIMAL chegam como texto
// e são convertidos para centavos em utils/dinheiro.js, sem perder precisão.
const pool = mysql.createPool({
  ...config.db,
  charset: "utf8mb4",
  dateStrings: true,
  waitForConnections: true,
  connectionLimit: 10,
});

// Executa uma função dentro de uma transação.
// Se a função lançar erro, tudo é desfeito (ROLLBACK).
const transacao = async (fn) => {
  const conexao = await pool.getConnection();

  try {
    await conexao.beginTransaction();
    const resultado = await fn(conexao);
    await conexao.commit();
    return resultado;
  } catch (erro) {
    await conexao.rollback();
    throw erro;
  } finally {
    conexao.release();
  }
};

module.exports = { pool, transacao };

// Validações do backend.
// As regras e as mensagens são as MESMAS do frontend (app/cadastro.js),
// e os erros usam os MESMOS nomes de campo (fullName, cpf, birthDate...),
// para que o app possa fazer setErrors(resposta.errors) diretamente.

const MENSAGENS = {
  fullName: "Informe seu nome completo.",
  cpf: "Informe um CPF válido.",
  birthDate: "Informe uma data válida e tenha pelo menos 18 anos.",
  email: "Informe um e-mail válido.",
  phone: "Informe um telefone válido.",
  password:
    "A senha deve ter 8–64 caracteres, com maiúscula, minúscula, número e caractere especial.",
  confirmPassword: "As senhas não são iguais.",
  pin: "O PIN deve ter 6 dígitos e não pode ser uma sequência ou repetição.",
  confirmPin: "Os PINs não são iguais.",
  terms: "Aceite os termos para continuar.",
};

// Garante que o valor recebido seja texto (o JSON pode trazer qualquer tipo).
const texto = (valor) => (typeof valor === "string" ? valor : "");

// Remove caracteres que não sejam números.
const onlyNumbers = (value) => texto(value).replace(/\D/g, "");

// Valida o CPF utilizando os dois dígitos verificadores.
const validateCPF = (value) => {
  const numbers = onlyNumbers(value);

  if (numbers.length !== 11) {
    return false;
  }

  if (/^(\d)\1{10}$/.test(numbers)) {
    return false;
  }

  let sum = 0;

  for (let i = 0; i < 9; i++) {
    sum += Number(numbers[i]) * (10 - i);
  }

  let remainder = (sum * 10) % 11;

  if (remainder === 10) {
    remainder = 0;
  }

  if (remainder !== Number(numbers[9])) {
    return false;
  }

  sum = 0;

  for (let i = 0; i < 10; i++) {
    sum += Number(numbers[i]) * (11 - i);
  }

  remainder = (sum * 10) % 11;

  if (remainder === 10) {
    remainder = 0;
  }

  return remainder === Number(numbers[10]);
};

// Valida o formato básico do e-mail.
const validateEmail = (value) => {
  const email = texto(value);
  return email.length <= 150 && /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email);
};

// Converte "DD-MM-AAAA" (formato do app) para "AAAA-MM-DD" (formato do MySQL).
// Retorna null se a data não existir.
const birthDateToSql = (value) => {
  const data = texto(value);

  if (!/^\d{2}-\d{2}-\d{4}$/.test(data)) {
    return null;
  }

  const [day, month, year] = data.split("-").map(Number);
  const date = new Date(year, month - 1, day);

  if (
    date.getFullYear() !== year ||
    date.getMonth() + 1 !== month ||
    date.getDate() !== day
  ) {
    return null;
  }

  return `${year}-${String(month).padStart(2, "0")}-${String(day).padStart(2, "0")}`;
};

// Valida a data e verifica a idade mínima de 18 anos.
const validateBirthDate = (value) => {
  const sql = birthDateToSql(value);

  if (!sql) {
    return false;
  }

  const [year, month, day] = sql.split("-").map(Number);
  const today = new Date();
  let age = today.getFullYear() - year;

  const hasBirthdayPassed =
    today.getMonth() + 1 > month ||
    (today.getMonth() + 1 === month && today.getDate() >= day);

  if (!hasBirthdayPassed) {
    age--;
  }

  return age >= 18 && year >= 1900;
};

// Valida telefone brasileiro com 10 ou 11 dígitos.
const validatePhone = (value) => {
  const numbers = onlyNumbers(value);
  return numbers.length === 10 || numbers.length === 11;
};

// Valida as regras definidas para a senha.
const validatePassword = (value) => {
  const senha = texto(value);

  if (senha.length < 8 || senha.length > 64) {
    return false;
  }

  return (
    /[A-Z]/.test(senha) &&
    /[a-z]/.test(senha) &&
    /\d/.test(senha) &&
    /[^A-Za-z0-9]/.test(senha)
  );
};

// Valida as regras básicas do PIN.
const validatePin = (value) => {
  const pin = texto(value);

  if (!/^\d{6}$/.test(pin)) {
    return false;
  }

  if (/^(\d)\1{5}$/.test(pin)) {
    return false;
  }

  return pin !== "123456" && pin !== "654321";
};

// Valida todos os campos da tela de cadastro.
// Retorna um objeto { campo: mensagem } — vazio quando está tudo certo.
const validarCadastro = (body) => {
  const errors = {};
  const fullName = texto(body.fullName).trim();

  if (fullName.length < 3 || fullName.length > 150) {
    errors.fullName = MENSAGENS.fullName;
  }

  if (!validateCPF(body.cpf)) {
    errors.cpf = MENSAGENS.cpf;
  }

  if (!validateBirthDate(body.birthDate)) {
    errors.birthDate = MENSAGENS.birthDate;
  }

  if (!validateEmail(texto(body.email).trim())) {
    errors.email = MENSAGENS.email;
  }

  if (!validatePhone(body.phone)) {
    errors.phone = MENSAGENS.phone;
  }

  if (!validatePassword(body.password)) {
    errors.password = MENSAGENS.password;
  }

  if (texto(body.password) !== texto(body.confirmPassword)) {
    errors.confirmPassword = MENSAGENS.confirmPassword;
  }

  if (!validatePin(body.pin)) {
    errors.pin = MENSAGENS.pin;
  }

  if (texto(body.pin) !== texto(body.confirmPin)) {
    errors.confirmPin = MENSAGENS.confirmPin;
  }

  if (body.termsAccepted !== true) {
    errors.terms = MENSAGENS.terms;
  }

  return errors;
};

module.exports = {
  MENSAGENS,
  texto,
  onlyNumbers,
  validateCPF,
  validateEmail,
  validateBirthDate,
  birthDateToSql,
  validatePhone,
  validatePassword,
  validatePin,
  validarCadastro,
};

const { pool } = require("../db");
const { hashToken } = require("../seguranca/criptografia");
const { ErroApi } = require("../utils/erros");

// Protege as rotas que exigem login.
// O app deve enviar o header:  Authorization: Bearer <token>
const autenticar = async (req, res, next) => {
  const [tipo, token] = (req.get("authorization") || "").split(" ");

  if (tipo !== "Bearer" || !token) {
    throw new ErroApi(401, "Faça login para continuar.");
  }

  // O banco guarda apenas o hash do token.
  const [sessoes] = await pool.execute(
    `SELECT s.id AS sessao_id, s.cliente_id, s.expiracao <= NOW() AS expirada,
            cl.status AS cliente_status, ua.id AS usuario_id,
            ua.status AS usuario_status
       FROM sessao s
       JOIN cliente cl ON cl.id = s.cliente_id
       JOIN usuario_acesso ua ON ua.cliente_id = cl.id
      WHERE s.token_hash = ? AND s.status = 'ATIVA'`,
    [hashToken(token)],
  );

  if (sessoes.length === 0) {
    throw new ErroApi(401, "Sessão inválida. Faça login novamente.");
  }

  const sessao = sessoes[0];

  if (Number(sessao.expirada) === 1) {
    await pool.execute("UPDATE sessao SET status = 'EXPIRADA' WHERE id = ?", [
      sessao.sessao_id,
    ]);
    throw new ErroApi(401, "Sessão expirada. Faça login novamente.");
  }

  if (sessao.cliente_status !== "ATIVO" || sessao.usuario_status !== "ATIVO") {
    throw new ErroApi(403, "Acesso não permitido. Entre em contato com o banco.");
  }

  req.auth = {
    sessaoId: Number(sessao.sessao_id),
    clienteId: Number(sessao.cliente_id),
    usuarioId: Number(sessao.usuario_id),
  };

  next();
};

module.exports = { autenticar };

// Rota inexistente.
const rotaNaoEncontrada = (req, res) => {
  res.status(404).json({ message: "Rota não encontrada." });
};

// Tratamento central de erros. Sempre responde em JSON.
// eslint-disable-next-line no-unused-vars
const tratarErros = (erro, req, res, next) => {
  if (erro instanceof ErroApi) {
    const corpo = { message: erro.message };

    if (erro.errors) {
      corpo.errors = erro.errors;
    }

    return res.status(erro.status).json(corpo);
  }

  if (erro.type === "entity.parse.failed") {
    return res.status(400).json({ message: "JSON inválido." });
  }

  if (erro.type === "entity.too.large") {
    return res.status(413).json({ message: "Requisição muito grande." });
  }

  console.error("[erro]", erro);
  return res
    .status(500)
    .json({ message: "Erro interno. Tente novamente mais tarde." });
};

module.exports = { rotaNaoEncontrada, tratarErros };

router.post("/cadastro", async (req, res) => {
  const body = req.body;
  const errors = validarCadastro(body);

  if (Object.keys(errors).length > 0) {
    throw new ErroApi(400, "Verifique os dados informados.", errors);
  }

  // Normaliza os dados para o formato das colunas do banco.
  const dados = {
    nome: texto(body.fullName).trim(),
    cpf: onlyNumbers(body.cpf),
    nascimento: birthDateToSql(body.birthDate),
    email: texto(body.email).trim().toLowerCase(),
    telefone: onlyNumbers(body.phone),
  };

  // Verifica se CPF ou e-mail já existem.
  const [existentes] = await pool.execute(
    "SELECT cpf, email FROM cliente WHERE cpf = ? OR email = ?",
    [dados.cpf, dados.email],
  );
  const duplicados = {};

  for (const cliente of existentes) {
    if (cliente.cpf === dados.cpf) {
      duplicados.cpf = "Este CPF já está cadastrado.";
    }
    if (cliente.email === dados.email) {
      duplicados.email = "Este e-mail já está cadastrado.";
    }
  }

  if (Object.keys(duplicados).length > 0) {
    throw new ErroApi(409, "Verifique os dados informados.", duplicados);
  }

  // A senha é guardada somente como hash bcrypt.
  // O PIN é validado, mas NÃO é gravado: o banco não possui coluna para ele.
  const senhaHash = await gerarHashSenha(body.password);

  let resultado;

  try {
    resultado = await transacao(async (conexao) => {
      const [cliente] = await conexao.execute(
        `INSERT INTO cliente (cpf, nome_completo, data_nascimento, email, telefone)
         VALUES (?, ?, ?, ?, ?)`,
        [dados.cpf, dados.nome, dados.nascimento, dados.email, dados.telefone],
      );
      const clienteId = cliente.insertId;

      const [usuario] = await conexao.execute(
        "INSERT INTO usuario_acesso (cliente_id, senha_hash) VALUES (?, ?)",
        [clienteId, senhaHash],
      );

      const contaId = await abrirConta(conexao);

      await conexao.execute(
        `INSERT INTO titularidade (conta_id, cliente_id, tipo_titular)
         VALUES (?, ?, 'TITULAR')`,
        [contaId, clienteId],
      );

      // O aceite dos termos fica registrado na auditoria.
      for (const acao of ["CADASTRO", "ACEITE_TERMOS"]) {
        await registrarAuditoria(
          {
            clienteId,
            usuarioId: usuario.insertId,
            acao,
            entidade: "cliente",
            entidadeId: clienteId,
            ip: req.ip,
            resultado: "SUCESSO",
          },
          conexao,
        );
      }

      const [[clienteCriado]] = await conexao.execute(
        "SELECT * FROM cliente WHERE id = ?",
        [clienteId],
      );
      const [conta] = await listarContas(clienteId, conexao);

      return { cliente: clienteCriado, conta };
    });
  } catch (erro) {
    // Dois cadastros iguais ao mesmo tempo: o UNIQUE do banco barra o segundo.
    if (erro.code === "ER_DUP_ENTRY") {
      throw new ErroApi(409, "Este CPF ou e-mail já está cadastrado.");
    }
    throw erro;

  res.status(201).json({
    message: "Conta criada com sucesso.",
    cliente: formatarCliente(resultado.cliente),
    conta: formatarConta(resultado.conta),
  });
});

const MAX_CHAVES_ALEATORIAS = 5;
const UUID_REGEX =
  /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;

// Formas possíveis de uma chave digitada pelo usuário, na ordem de busca.
// Ex.: "12999990001" pode ser um CPF ou um telefone (+5512999990001).
const formasDaChave = (chave) => {
  if (chave.includes("@")) {
    return [chave.toLowerCase()];
  }

  if (UUID_REGEX.test(chave)) {
    return [chave.toLowerCase()];
  }

  const numeros = onlyNumbers(chave);

  if (chave.startsWith("+")) {
    return [`+${numeros}`];
  }

  if (numeros.length === 11) {
    return [numeros, `+55${numeros}`];
  }

  if (numeros.length === 10) {
    return [`+55${numeros}`];
  }

  if (numeros.length >= 12 && numeros.startsWith("55")) {
    return [`+${numeros}`];
  }

  return [chave];
};

router.post("/chaves", async (req, res) => {

  if (!TIPOS_CHAVE.includes(tipo)) {
    throw new ErroApi(400, "Verifique os dados informados.", {
      keyType: "Tipo de chave inválido. Use CPF, EMAIL, TELEFONE ou ALEATORIA.",

    clienteId,
  ]);

  let valor;

  if (tipo === "CPF") {

  } else if (tipo === "EMAIL") {
    valor = cliente.email;

      [clienteId],

        keyType: `Limite de ${MAX_CHAVES_ALEATORIAS} chaves aleatórias atingido.`,
      });

    valor = gerarUuid();

  const [existentes] = await pool.execute(
    "SELECT * FROM chave_pix WHERE chave = ?",

  if (existentes.length > 0) {

    const ehMinhaInativa =

          keyType: "Esta chave Pix já está cadastrada.",
        });

    entidadeId: chaveId,

// Body: {

router.post("/enviar", async (req, res) => {
  const body = req.body;
  const clienteId = req.auth.clienteId;
  const usuarioId = req.auth.usuarioId;
  const errors = {};

  const chaveDigitada = texto(body.key).trim();
  const centavos = lerValor(body.amount);
  const limiteCentavos = Math.round(config.pixLimitePorTransacao * 100);
  const descricaoValida =
    body.description === undefined ||
    body.description === null ||
    typeof body.description === "string";
  const descricao = texto(body.description).trim();

  if (!chaveDigitada || chaveDigitada.length > 255) {
    errors.key = "Informe a chave Pix.";
  }

  if (centavos === null || centavos <= 0) {
    errors.amount = "Informe um valor válido.";
  } else if (centavos > limiteCentavos) {
    errors.amount = `O valor máximo por Pix é R$ ${paraDecimal(limiteCentavos).replace(".", ",")}.`;
  }

  if (!descricaoValida || descricao.length > 140) {
    errors.description = "A descrição deve ter no máximo 140 caracteres.";
  }

  if (!texto(body.password)) {
    errors.password = "Informe sua senha.";
  }

  if (Object.keys(errors).length > 0) {
    throw new ErroApi(400, "Verifique os dados informados.", errors);
  }

  const auditoriaBase = {
    clienteId,
    usuarioId,
    acao: "PIX_ENVIADO",
    entidade: "pix",
    ip: req.ip,
  };

  // 1) Confirma a senha (mesmo contador de tentativas do login).
  const [[usuario]] = await pool.execute(
    `SELECT senha_hash,
            bloqueado_ate IS NOT NULL AND bloqueado_ate > NOW() AS bloqueado
       FROM usuario_acesso WHERE id = ?`,
    [usuarioId],
  );

  if (Number(usuario.bloqueado) === 1) {
    throw new ErroApi(
      423,
      "Acesso bloqueado temporariamente por excesso de tentativas. Tente novamente mais tarde.",
    );
  }

  if (!(await conferirSenha(body.password, usuario.senha_hash))) {
    const bloqueou = await registrarSenhaErrada(usuarioId);
    await registrarAuditoria({ ...auditoriaBase, resultado: "FALHA" });

    if (bloqueou) {
      throw new ErroApi(
        423,
        "Acesso bloqueado temporariamente por excesso de tentativas. Tente novamente mais tarde.",
      );
    }
    throw new ErroApi(401, "Senha incorreta.", { password: "Senha incorreta." });
  }

  await limparTentativas(usuarioId);

  // 2) Encontra a conta de destino pela chave Pix.
  const formas = formasDaChave(chaveDigitada);
  const [encontradas] = await pool.execute(
    `SELECT cp.chave, cp.conta_id, cl.nome_completo,
            c.numero_conta, c.digito, a.codigo_agencia, b.codigo_banco
       FROM chave_pix cp
       JOIN cliente cl ON cl.id = cp.cliente_id
       JOIN conta c ON c.id = cp.conta_id
       JOIN agencia a ON a.id = c.agencia_id
       JOIN banco b ON b.id = a.banco_id
      WHERE cp.status = 'ATIVA' AND cp.chave IN (${formas.map(() => "?").join(", ")})`,
    formas,
  );
  const destino = formas
    .map((forma) => encontradas.find((linha) => linha.chave === forma))
    .find(Boolean);

  if (!destino) {
    await registrarAuditoria({ ...auditoriaBase, resultado: "FALHA" });
    throw new ErroApi(404, "Chave Pix não encontrada.", {
      key: "Chave Pix não encontrada.",
    });
  }

  // 3) Conta de origem (precisa pertencer ao cliente logado).
  const origem = await resolverContaOrigem(body.accountId, clienteId);
  const destinoId = Number(destino.conta_id);

      key: "Não é possível enviar um Pix para a mesma conta.",
    });
  }

  const [[remetente]] = await pool.execute(
    "SELECT nome_completo FROM cliente WHERE id = ?",
    [clienteId],
  );

  // 4) Movimenta o dinheiro numa única transação do banco.
  //    As duas contas são travadas (FOR UPDATE) sempre na mesma ordem (por id)
  //    para evitar saldo duplicado e deadlock com Pix simultâneos.
  let pix;

  try {
    pix = await transacao(async (conexao) => {
      const ids = [origemId, destinoId].sort((a, b) => a - b);
      const [travadas] = await conexao.execute(

           FROM conta WHERE id IN (?, ?) ORDER BY id FOR UPDATE`,
        ids,
      );

      const contaDestino = travadas.find((c) => Number(c.id) === destinoId);

        throw new ErroApi(422, "A conta de origem está bloqueada ou encerrada.");
      }

      if (contaDestino.status !== "ATIVA") {
        throw new ErroApi(422, "Verifique os dados informados.", {
          key: "A conta de destino não pode receber Pix.",
        });
      }

      const saldoOrigem = paraCentavos(contaOrigem.saldo);
      const limiteOrigem = paraCentavos(contaOrigem.limite_cheque_especial);
      const novoSaldoOrigem = saldoOrigem - centavos;

      if (novoSaldoOrigem < -limiteOrigem) {
        throw new ErroApi(422, "Saldo insuficiente.", {
          amount: "Saldo insuficiente.",
        });
      }

      const saldoDestino = paraCentavos(contaDestino.saldo);
      const novoSaldoDestino = saldoDestino + centavos;
      const valor = paraDecimal(centavos);
      const endToEndId = gerarEndToEndId();

      const [novoPix] = await conexao.execute(
        `INSERT INTO pix
           (conta_origem_id, conta_destino_id, chave_destino, valor, descricao, end_to_end_id)
         VALUES (?, ?, ?, ?, ?, ?)`,

      );

        `INSERT INTO transacao
           (conta_id, tipo, valor, descricao, saldo_anterior, saldo_posterior, identificador)
         VALUES (?, 'PIX_ENVIADO', ?, ?, ?, ?, ?)`,
        [
          origemId,
          valor,
          (descricao || `Pix enviado para ${destino.nome_completo}`).slice(0, 255),
          paraDecimal(saldoOrigem),
          paraDecimal(novoSaldoOrigem),
          gerarUuid(),
        ],
      );

      await conexao.execute("UPDATE conta SET saldo = ? WHERE id = ?", [
        paraDecimal(novoSaldoOrigem),
        origemId,
      ]);

      const [entrada] = await conexao.execute(

           (conta_id, tipo, valor, descricao, saldo_anterior, saldo_posterior, identificador)
         VALUES (?, 'PIX_RECEBIDO', ?, ?, ?, ?, ?)`,
        [
          destinoId,
          valor,
          (descricao || `Pix recebido de ${remetente.nome_completo}`).slice(0, 255),
          paraDecimal(saldoDestino),
          paraDecimal(novoSaldoDestino),
          gerarUuid(),
        ],
      );

      await conexao.execute("UPDATE conta SET saldo = ? WHERE id = ?", [

    }
    throw erro;
  }

  const contaAtualizada = await buscarContaDoCliente(origemId, clienteId);

const crypto = require("node:crypto");
const bcrypt = require("bcryptjs");

const config = require("../config");

// Hash usado quando o usuário não existe, para que o tempo de resposta
// do login seja parecido e não revele se o e-mail/CPF está cadastrado.
const HASH_FALSO = bcrypt.hashSync("parabon-hash-falso", 10);

// Gera o hash bcrypt da senha (gravado em usuario_acesso.senha_hash).
const gerarHashSenha = (senha) => {
  return bcrypt.hash(senha, config.bcryptCusto);
};

// Compara a senha digitada com o hash gravado no banco.
// Hashes inválidos (como os de exemplo do script SQL) retornam false.
const conferirSenha = async (senha, hash) => {
  try {
    return await bcrypt.compare(senha, hash || HASH_FALSO);
  } catch {
    return false;
  }
};

// Gera o token de sessão entregue ao aplicativo.
// 32 bytes aleatórios = 256 bits, em base64url (seguro para URL e header).
const gerarTokenSessao = () => {
  return crypto.randomBytes(32).toString("base64url");
};

// O banco guarda somente o SHA-256 do token (sessao.token_hash),
// nunca o token em si. Se o banco vazar, os tokens não podem ser usados.
const hashToken = (token) => {
  return crypto.createHash("sha256").update(token).digest("hex");
};

// Identificador único para transacao.identificador e chaves aleatórias.
const gerarUuid = () => crypto.randomUUID();

// Identificador do Pix no padrão E + ISPB(8) + AAAAMMDDHHmm + 11 caracteres.
const gerarEndToEndId = () => {
  const ispb = String(config.bancoCodigo).padStart(8, "0").slice(-8);
  const agora = new Date();
  const data = [
    agora.getUTCFullYear(),
    String(agora.getUTCMonth() + 1).padStart(2, "0"),
    String(agora.getUTCDate()).padStart(2, "0"),
    String(agora.getUTCHours()).padStart(2, "0"),
    String(agora.getUTCMinutes()).padStart(2, "0"),
  ].join("");
  const alfabeto = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789";
  const bytes = crypto.randomBytes(11);
  const sufixo = Array.from(bytes, (b) => alfabeto[b % alfabeto.length]).join(
    "",
  );

  return `E${ispb}${data}${sufixo}`;
};

module.exports = {
  gerarHashSenha,
  conferirSenha,
  gerarTokenSessao,
  hashToken,
  gerarUuid,
  gerarEndToEndId,
};

const crypto = require("node:crypto");

const config = require("../config");
const { pool } = require("../db");

// Consulta base das contas de um cliente (somente titularidades vigentes).
const SELECT_CONTAS = `
  SELECT c.*, a.codigo_agencia, a.digito AS digito_agencia,
         b.codigo_banco, t.tipo_titular
    FROM conta c
    JOIN agencia a ON a.id = c.agencia_id
    JOIN banco b ON b.id = a.banco_id
    JOIN titularidade t ON t.conta_id = c.id
   WHERE t.cliente_id = ? AND t.data_fim IS NULL`;

// Lista as contas do cliente. A conta em que ele é TITULAR vem primeiro.
const listarContas = async (clienteId, executor = pool) => {
  const [contas] = await executor.execute(
    `${SELECT_CONTAS}
     ORDER BY t.tipo_titular = 'TITULAR' DESC, c.id`,
    [clienteId],
  );

  return contas;
};

// Busca uma conta garantindo que ela pertence ao cliente.
// Responde 404 (e não 403) para não revelar contas de outras pessoas.
const buscarContaDoCliente = async (contaId, clienteId, executor = pool) => {
  const id = Number(contaId);

  if (!Number.isInteger(id) || id <= 0) {
    throw new ErroApi(404, "Conta não encontrada.");
  }

  const [contas] = await executor.execute(`${SELECT_CONTAS} AND c.id = ?`, [
    clienteId,
    id,
  ]);

  if (contas.length === 0) {
    throw new ErroApi(404, "Conta não encontrada.");
  }

  return contas[0];
};

// Retorna a conta informada ou, se nenhuma foi informada, a conta principal.
const resolverContaOrigem = async (contaId, clienteId, executor = pool) => {
  if (contaId !== undefined && contaId !== null && contaId !== "") {
    return buscarContaDoCliente(contaId, clienteId, executor);
  }

  const contas = await listarContas(clienteId, executor);

  if (contas.length === 0) {
    throw new ErroApi(404, "Nenhuma conta encontrada para este cliente.");
  }

  return contas[0];
};

// Busca a agência configurada (BANCO_CODIGO + AGENCIA_CODIGO no .env).
const buscarAgenciaPadrao = async (executor = pool) => {
  const [agencias] = await executor.execute(
    `SELECT a.id
       FROM agencia a
       JOIN banco b ON b.id = a.banco_id
      WHERE b.codigo_banco = ? AND a.codigo_agencia = ?
        AND a.status = 'ATIVA' AND b.status = 'ATIVO'`,
    [config.bancoCodigo, config.agenciaCodigo],
  );

  if (agencias.length === 0) {
    throw new Error(
      `Agência ${config.agenciaCodigo} do banco ${config.bancoCodigo} não encontrada. ` +
        "Confira BANCO_CODIGO e AGENCIA_CODIGO no .env.",
    );
  }

  return agencias[0].id;
};

// Dígito verificador da conta (módulo 11).
const calcularDigito = (numero) => {
  let soma = 0;
  let peso = 2;

  for (let i = numero.length - 1; i >= 0; i--) {
    soma += Number(numero[i]) * peso;
    peso = peso === 9 ? 2 : peso + 1;
  }

  const resto = 11 - (soma % 11);
  return resto >= 10 ? "0" : String(resto);
};

// Abre uma conta DIGITAL na agência padrão, com número aleatório de 8 dígitos.
const abrirConta = async (executor) => {
  const agenciaId = await buscarAgenciaPadrao(executor);

  for (let tentativa = 0; tentativa < 10; tentativa++) {
    const numero = String(crypto.randomInt(10000000, 100000000));
    const [existe] = await executor.execute(
      "SELECT id FROM conta WHERE agencia_id = ? AND numero_conta = ?",
      [agenciaId, numero],
    );

    if (existe.length > 0) {
      continue;
    }

    const [resultado] = await executor.execute(
      `INSERT INTO conta (agencia_id, numero_conta, digito, tipo_conta)
       VALUES (?, ?, ?, 'DIGITAL')`,
      [agenciaId, numero, calcularDigito(numero)],
    );

    return resultado.insertId;
  }

  throw new Error("Não foi possível gerar um número de conta livre.");
};

  resolverContaOrigem,
  abrirConta,
  calcularDigito,
};

// Os cálculos de dinheiro são feitos em CENTAVOS (números inteiros)
// para evitar erros de arredondamento (ex.: 0.1 + 0.2 = 0.30000000000000004).

// Converte o DECIMAL do MySQL ("2500.00" ou "-10.5") para centavos.
const paraCentavos = (decimal) => {
  const [inteiro, fracao = ""] = String(decimal).replace("-", "").split(".");
  const centavos = Number(inteiro) * 100 + Number(fracao.padEnd(2, "0").slice(0, 2));
  return String(decimal).startsWith("-") ? -centavos : centavos;
};

// Converte centavos para o texto aceito pelo MySQL ("2500.00").
const paraDecimal = (centavos) => {
  const sinal = centavos < 0 ? "-" : "";
  const absoluto = Math.abs(centavos);
  return `${sinal}${Math.floor(absoluto / 100)}.${String(absoluto % 100).padStart(2, "0")}`;
};

// Converte centavos para número com 2 casas, usado nas respostas JSON.
const paraNumero = (centavos) => centavos / 100;

// Lê um valor enviado pelo app (10, 10.5, "10,50" ou "10.50").
// Retorna os centavos ou null se o valor for inválido.
const lerValor = (valor) => {
  let textoValor;

  if (typeof valor === "number" && Number.isFinite(valor)) {
    textoValor = valor.toFixed(2);
    if (Math.abs(Number(textoValor) - valor) > 1e-9) {
      return null; // mais de 2 casas decimais
    }
  } else if (typeof valor === "string") {
    textoValor = valor.trim().replace(",", ".");
  } else {
    return null;
  }

  if (!/^\d{1,13}(\.\d{1,2})?$/.test(textoValor)) {
    return null;
  }

  return paraCentavos(textoValor);
};

module.exports = { paraCentavos, paraDecimal, paraNumero, lerValor };

  if (!dataSql) {
    return null;
  }

  const [ano, mes, dia] = dataSql.slice(0, 10).split("-");

};
